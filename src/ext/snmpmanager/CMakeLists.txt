# 2 つの使い方がある
#   単体 : このディレクトリを cmake する。インストール済みの OpenRTM-aist (Linux: rtm2-config、Windows: OpenRTMConfig.cmake) を使う
#   拡張 : OpenRTM-aist (C++) のソースの src/ext/snmpmanager に置き、OpenRTM-aist と一緒にビルドする
#          (cmake -DSNMPMANAGER_ENABLE=ON。同じツリーの libRTC2・IDL・インストール先を使う)
if(COMMAND openrtm_include_rtm AND DEFINED RTM_PROJECT_NAME)
  set(SNMPMGR_IN_OPENRTM ON)
  message(STATUS "SNMPManager: built as an OpenRTM-aist extension (src/ext/snmpmanager)")
else()
  set(SNMPMGR_IN_OPENRTM OFF)
endif()
set(SRC ${CMAKE_CURRENT_SOURCE_DIR})
set(BIN ${CMAKE_CURRENT_BINARY_DIR})

set(CMAKE_CXX_STANDARD 17)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)

# ---- OpenRTM-aist ------------------------------------------------------------------------------
if(SNMPMGR_IN_OPENRTM)
  # OpenRTM-aist のツリーの中: 同じツリーでビルドする libRTC2 と IDL を使う
  link_directories(${ORB_LINK_DIR})
  add_definitions(${ORB_C_FLAGS_LIST})
  add_definitions(${COIL_C_FLAGS_LIST})
  set(RTM_DEFINITIONS)
  set(RTM_INCLUDE_DIRS)
  set(RTM_LIBRARIES ${RTM_PROJECT_NAME} ${ORB_LIBRARIES} ${RTM_LINKER_OPTION} Threads::Threads)
  if(WIN32)
    add_definitions(-DRTM_SKEL_IMPORT_SYMBOL)
    list(APPEND RTM_LIBRARIES ${RTCSKEL_PROJECT_NAME})
  else()
    list(APPEND RTM_LIBRARIES dl)
  endif()
  set(OMNIIDL ${IDL_COMPILE_COMMAND})
  set(RTM_IDL_DIR ${CMAKE_SOURCE_DIR}/src/lib/rtm/idl)
  # インストール先 (OpenRTM-aist と同じ)
  set(SNMPMGR_INSTALL_BIN ${INSTALL_RTM_BIN_DIR})
  set(SNMPMGR_INSTALL_EXT ${INSTALL_RTM_EXT_DIR}/snmpmanager)
  set(SNMPMGR_INSTALL_INCLUDE ${INSTALL_RTM_INCLUDE_DIR})
elseif(WIN32)
  # 単体 (Windows): OpenRTM-aist 2.1.0 (C++、vc16) のインストール先 (OpenRTMConfig.cmake)
  set(OpenRTM_DIR "C:/Program Files/OpenRTM-aist/2.1.0/cmake" CACHE PATH "OpenRTMConfig.cmake のディレクトリ")
  find_package(OpenRTM REQUIRED)
  # OpenRTM の _WIN32_WINNT (0x0500 と 0x0603 の 2 つが入っている) は、Windows 10 のものに置き換える
  set(RTM_DEFINITIONS)
  foreach(f ${OPENRTM_CFLAGS})
    if(NOT f MATCHES "^-D_WIN32_WINNT=")
      list(APPEND RTM_DEFINITIONS ${f})
    endif()
  endforeach()
  list(APPEND RTM_DEFINITIONS -D_WIN32_WINNT=0x0A00)
  set(RTM_INCLUDE_DIRS ${OPENRTM_INCLUDE_DIRS})
  link_directories(${OPENRTM_LIBRARY_DIRS})
  set(RTM_LIBRARIES ${OPENRTM_LIBRARIES})
  set(OMNIIDL_PATH "${OMNIORB_DIR}/bin/x86_win32/omniidl.exe")
  set(RTM_IDL_PATH "${OPENRTM_DIR}/rtm/idl")
  include(GNUInstallDirs)
  set(SNMPMGR_INSTALL_BIN ${CMAKE_INSTALL_BINDIR})
  set(SNMPMGR_INSTALL_EXT ${CMAKE_INSTALL_LIBDIR}/snmpmanager)
  set(SNMPMGR_INSTALL_INCLUDE ${CMAKE_INSTALL_INCLUDEDIR})
else()
  # 単体 (Linux): rtm2-config から include ディレクトリを取る
  execute_process(COMMAND rtm2-config --cflags OUTPUT_VARIABLE RTM_CFLAGS OUTPUT_STRIP_TRAILING_WHITESPACE)
  string(REGEX MATCHALL "-I[^ ]+" RTM_INC_FLAGS "${RTM_CFLAGS}")
  set(RTM_INCLUDE_DIRS "")
  foreach(f ${RTM_INC_FLAGS})
    string(SUBSTRING "${f}" 2 -1 d)
    list(APPEND RTM_INCLUDE_DIRS "${d}")
  endforeach()
  set(RTM_DEFINITIONS)
  set(RTM_LIBRARIES RTC2 omniDynamic4 omniORB4 omnithread Threads::Threads dl)
  set(OMNIIDL omniidl)
  set(RTM_IDL_DIR /usr/include/openrtm-2.1/rtm/idl)
  include(GNUInstallDirs)
  set(SNMPMGR_INSTALL_BIN ${CMAKE_INSTALL_BINDIR})
  set(SNMPMGR_INSTALL_EXT ${CMAKE_INSTALL_LIBDIR}/snmpmanager)
  set(SNMPMGR_INSTALL_INCLUDE ${CMAKE_INSTALL_INCLUDEDIR})
endif()

# ---- そのほかの依存 --------------------------------------------------------------------------------
if(WIN32)
  if(SNMPMGR_IN_OPENRTM)
    set(OMNIIDL_PATH "${OMNIIDL}")
    set(RTM_IDL_PATH "${RTM_IDL_DIR}")
  endif()
  # omniidl はプリプロセッサのパスを引用符なしで呼ぶので、空白のない短い名前 (8.3 形式) にする
  file(WRITE ${BIN}/shortpath.bat "@echo %~s1\n")
  function(short_path in out)
    file(TO_NATIVE_PATH "${in}" native)
    execute_process(COMMAND ${BIN}/shortpath.bat "${native}" OUTPUT_VARIABLE s OUTPUT_STRIP_TRAILING_WHITESPACE)
    set(${out} "${s}" PARENT_SCOPE)
  endfunction()
  if(EXISTS "${OMNIIDL_PATH}")
    short_path("${OMNIIDL_PATH}" OMNIIDL)
  else()
    set(OMNIIDL "${OMNIIDL_PATH}")
  endif()
  short_path("${RTM_IDL_PATH}" RTM_IDL_DIR)
  # nlohmann_json: 見つからなければ同梱 (third_party/nlohmann。Debian の nlohmann-json3-dev 3.11.3 のヘッダ)
  find_package(nlohmann_json QUIET)
  if(NOT nlohmann_json_FOUND)
    add_library(snmpmgr_nlohmann_json INTERFACE)
    target_include_directories(snmpmgr_nlohmann_json INTERFACE ${SRC}/third_party)
    add_library(nlohmann_json::nlohmann_json ALIAS snmpmgr_nlohmann_json)
  endif()
  # tinyxml2: CMAKE_PREFIX_PATH か TINYXML2_DIR (既定は Chocolatey のパッケージ)
  set(TINYXML2_DIR "C:/ProgramData/chocolatey/lib/tinyxml2" CACHE PATH "tinyxml2 の include / lib のある場所")
  find_path(TINYXML2_INCLUDE_DIRS tinyxml2.h HINTS ${TINYXML2_DIR}/include REQUIRED)
  find_library(TINYXML2_LIBRARIES tinyxml2 HINTS ${TINYXML2_DIR}/lib REQUIRED)
  find_file(TINYXML2_DLL tinyxml2.dll HINTS ${TINYXML2_DIR}/lib ${TINYXML2_DIR}/bin)
  # SQLite: Windows SDK の winsqlite3 (OS の winsqlite3.dll を使う)
  set(SQLITE_INCLUDE_DIRS "")
  set(SQLITE_LIBRARIES winsqlite3)
  set(SQLITE_DEFINITIONS SNMPMGR_WINSQLITE)
  set(PLATFORM_LIBRARIES ws2_32 iphlpapi psapi)
  add_compile_options(/utf-8 /EHsc /bigobj /W3 /wd4996 /wd4251 /wd4275 /Zc:__cplusplus)
  # Release でも PDB とマップファイルを作る (落ちたときに場所を調べられるように)
  add_compile_options(/Zi)
  add_link_options(/DEBUG /OPT:REF /OPT:ICF /MAP)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
  # 実行ファイル・DLL は bin/ に、プラグインは bin/plugins/ に (構成ごとのサブディレクトリを作らない)
  set(OUT_DIR "$<1:${BIN}/bin>")
  set(PLUGIN_OUT_DIR "$<1:${BIN}/bin/plugins>")
  set(PLUGIN_TEST_DIR "${BIN}/bin/plugins")
  # 実行ファイルの ANSI コードページを UTF-8 にする (日本語を含むパスを std::string のまま扱うため)
  set(EXE_MANIFEST ${SRC}/win/utf8.manifest)
else()
  find_package(nlohmann_json REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SQLITE REQUIRED sqlite3)
  pkg_check_modules(TINYXML2 REQUIRED tinyxml2)
  set(SQLITE_DEFINITIONS)
  set(PLATFORM_LIBRARIES dl)
  add_compile_options(-Wall -Wno-unused-parameter -Wno-deprecated-declarations)
  set(OUT_DIR ${BIN})
  set(PLUGIN_OUT_DIR ${BIN}/plugins)
  set(PLUGIN_TEST_DIR ${BIN}/plugins)
  set(EXE_MANIFEST)
endif()

# コンポーネントオブザーバーのスケルトン (omniidl で生成。OpenRTM 版と FSM4RTC (omg.org/RTC) 版)
set(OBSERVER_GEN ${BIN}/idl)
file(MAKE_DIRECTORY ${OBSERVER_GEN})
set(OBSERVER_SRCS)
# (入力は生成先からの相対パスで渡す。Windows の omniidl は日本語を含む絶対パスを読めない)
file(RELATIVE_PATH IDL_REL ${OBSERVER_GEN} ${SRC}/idl)
foreach(idl ComponentObserver_OpenRTM ComponentObserver_RTC)
  add_custom_command(
    OUTPUT ${OBSERVER_GEN}/${idl}.hh ${OBSERVER_GEN}/${idl}SK.cc
    COMMAND ${OMNIIDL} -bcxx -Wbh=.hh -Wbs=SK.cc -I${RTM_IDL_DIR} ${IDL_REL}/${idl}.idl
    WORKING_DIRECTORY ${OBSERVER_GEN}
    DEPENDS ${SRC}/idl/${idl}.idl)
  # 生成したスタブは、包むファイルからコンパイルする。Windows では、OpenRTM の DLL にある SDOPackage のスタブ
  # (リポジトリ ID などのデータ) を dllimport で参照するように、先に USE_stub_in_nt_dll を付けて読む
  # (付けないと、プロキシの is_a() が DLL のデータを読めずに落ちる)
  file(WRITE ${OBSERVER_GEN}/${idl}_stub.cpp
       "// 生成ファイル (CMakeLists.txt)\n"
       "#if defined(_WIN32)\n#  define USE_stub_in_nt_dll\n#  include <SDOPackage.hh>\n#  undef USE_stub_in_nt_dll\n#endif\n"
       "#include \"${idl}SK.cc\"\n")
  set_source_files_properties(${OBSERVER_GEN}/${idl}_stub.cpp PROPERTIES OBJECT_DEPENDS ${OBSERVER_GEN}/${idl}SK.cc)
  list(APPEND OBSERVER_SRCS ${OBSERVER_GEN}/${idl}_stub.cpp ${OBSERVER_GEN}/${idl}.hh)
endforeach()

# ILP のソルバー (GLPK) の配置プラグイン。GLPK があるときだけ作る
#   Linux: apt の libglpk-dev。Windows: GLPK_DIR に winglpk の展開先 (glpk.h と glpk_*.lib / .dll のある場所)
set(GLPK_DIR_DEFAULT "")
if(WIN32)
  file(TO_CMAKE_PATH "$ENV{LOCALAPPDATA}/glpk/glpk-4.65" GLPK_DIR_DEFAULT)   # winglpk 4.65 を展開した場所 (既定)
endif()
set(GLPK_DIR "${GLPK_DIR_DEFAULT}" CACHE PATH "GLPK の場所 (glpk.h と glpk のライブラリ)")
find_path(GLPK_INCLUDE_DIR glpk.h HINTS ${GLPK_DIR} ${GLPK_DIR}/include ${GLPK_DIR}/src)
find_library(GLPK_LIBRARY NAMES glpk glpk_5_0 glpk_4_65
             HINTS ${GLPK_DIR} ${GLPK_DIR}/lib ${GLPK_DIR}/w64 ${GLPK_DIR}/lib/x64)
set(GLPK_DLL)
if(GLPK_INCLUDE_DIR AND GLPK_LIBRARY)
  message(STATUS "GLPK: ${GLPK_LIBRARY} (plugins/ILPPlacementPlugin.cpp)")
  add_library(ilp_placement MODULE plugins/ILPPlacementPlugin.cpp)
  target_include_directories(ilp_placement PRIVATE ${SRC}/include ${GLPK_INCLUDE_DIR})
  target_link_libraries(ilp_placement PRIVATE nlohmann_json::nlohmann_json ${GLPK_LIBRARY})
  set_target_properties(ilp_placement PROPERTIES PREFIX "lib" LIBRARY_OUTPUT_DIRECTORY ${PLUGIN_OUT_DIR})
  list(APPEND SNMPMGR_PLUGINS ilp_placement)
  if(SNMPMANAGER_BUILD_TESTS)
    add_dependencies(snmpmgr_tests ilp_placement)
    target_compile_definitions(snmpmgr_tests PRIVATE SNMPMGR_HAVE_ILP_PLUGIN)
  endif()
  if(WIN32)
    # glpk の DLL をプラグインの隣 (bin/plugins) に置く。プラグインは LoadLibraryEx(LOAD_WITH_ALTERED_SEARCH_PATH) で
    # 読むので、依存する DLL はプラグインのディレクトリから探される (実行ファイルのディレクトリは探されない)
    get_filename_component(GLPK_LIB_DIR ${GLPK_LIBRARY} DIRECTORY)
    get_filename_component(GLPK_LIB_NAME ${GLPK_LIBRARY} NAME_WE)
    if(EXISTS ${GLPK_LIB_DIR}/${GLPK_LIB_NAME}.dll)
      set(GLPK_DLL ${GLPK_LIB_DIR}/${GLPK_LIB_NAME}.dll)
      add_custom_command(TARGET ilp_placement POST_BUILD
                         COMMAND ${CMAKE_COMMAND} -E copy_if_different ${GLPK_DLL} ${PLUGIN_TEST_DIR}/)
    endif()
  endif()
else()
  message(STATUS "GLPK not found: plugins/ILPPlacementPlugin.cpp is not built")
endif()
