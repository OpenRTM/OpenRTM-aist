set(CSP_SAMPLE_SUPPORT_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE INTERNAL "CSP sample support source")
add_library(csp_sample_support STATIC SampleSupport.cpp)
openrtm_common_set_compile_props(csp_sample_support)
openrtm_include_rtm(csp_sample_support)
target_include_directories(csp_sample_support PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(csp_sample_support PRIVATE ${ORB_C_FLAGS_LIST} ${COIL_C_FLAGS_LIST})
if(WIN32)
  target_compile_definitions(csp_sample_support PRIVATE RTM_SKEL_IMPORT_SYMBOL)
endif()

function(csp_sample_settings target)
  openrtm_common_set_compile_props(${target})
  openrtm_include_rtm(${target})
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CSP_SAMPLE_SUPPORT_DIR}")
  target_compile_options(${target} PRIVATE ${ORB_C_FLAGS_LIST} ${COIL_C_FLAGS_LIST})
  if(WIN32)
    target_compile_definitions(${target} PRIVATE RTM_SKEL_IMPORT_SYMBOL)
  endif()
endfunction()

function(csp_sample_link target)
  target_link_directories(${target} PRIVATE ${ORB_LINK_DIR})
  target_link_libraries(${target} PRIVATE csp_sample_support ${RTM_PROJECT_NAME} ${ORB_LIBRARIES} ${RTM_LINKER_OPTION})
  if(WIN32)
    target_link_libraries(${target} PRIVATE ${RTCSKEL_PROJECT_NAME})
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:${RTM_PROJECT_NAME}> $<TARGET_FILE_DIR:${target}>
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:${RTCSKEL_PROJECT_NAME}> $<TARGET_FILE_DIR:${target}>)
    file(GLOB orb_dlls "${ORB_ROOT}/bin/x86_win32/*.dll")
    foreach(dll IN LISTS orb_dlls)
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" $<TARGET_FILE_DIR:${target}>)
    endforeach()
  endif()
endfunction()

function(csp_sample_component component)
  set(target "CSP${component}")
  add_library(${target}_objlib OBJECT ${component}.cpp ${component}.h ${ARGN})
  csp_sample_settings(${target}_objlib)
  add_library(${target} SHARED $<TARGET_OBJECTS:${target}_objlib>)
  set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "${component}" WINDOWS_EXPORT_ALL_SYMBOLS ON)
  csp_sample_link(${target})
  add_executable(${target}Comp "${CSP_SAMPLE_SUPPORT_DIR}/ComponentMain.cpp" $<TARGET_OBJECTS:${target}_objlib>)
  csp_sample_settings(${target}Comp)
  target_compile_definitions(${target}Comp PRIVATE "CSP_COMPONENT_INIT=${component}Init" "CSP_COMPONENT_TYPE=\"${component}\"")
  set_target_properties(${target}Comp PROPERTIES OUTPUT_NAME "${component}Comp")
  csp_sample_link(${target}Comp)
  set(CSP_SAMPLE_TARGETS ${CSP_SAMPLE_TARGETS} ${target}Comp PARENT_SCOPE)
  install(TARGETS ${target} ${target}Comp
    RUNTIME DESTINATION "${INSTALL_RTM_EXAMPLE_DIR}/CSP/${CSP_SAMPLE_GROUP}" COMPONENT examples
    LIBRARY DESTINATION "${INSTALL_RTM_EXAMPLE_DIR}/CSP/${CSP_SAMPLE_GROUP}" COMPONENT examples)
endfunction()

function(csp_sample_configs)
  file(GLOB configs "${CMAKE_CURRENT_SOURCE_DIR}/*.conf")
  file(GLOB launchers "${CMAKE_CURRENT_SOURCE_DIR}/*.ps1" "${CMAKE_CURRENT_SOURCE_DIR}/*.sh" "${CMAKE_CURRENT_SOURCE_DIR}/*.bat")
  foreach(config IN LISTS configs launchers)
    get_filename_component(name "${config}" NAME)
    configure_file("${config}" "${CMAKE_CURRENT_BINARY_DIR}/${name}" COPYONLY)
    foreach(target IN LISTS CSP_SAMPLE_TARGETS)
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${config}" "$<TARGET_FILE_DIR:${target}>/${name}")
    endforeach()
  endforeach()
  install(FILES ${configs}
    DESTINATION "${INSTALL_RTM_EXAMPLE_DIR}/CSP/${CSP_SAMPLE_GROUP}" COMPONENT examples)
endfunction()
